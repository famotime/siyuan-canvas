## 1. 统一强类型请求封装

```ts
import { fetchSyncPost, showMessage } from "siyuan";

export class ApiError extends Error {
  constructor(public code: number, message: string, public endpoint: string) {
    super(`[SiYuan API Error ${code}] ${endpoint}: ${message}`);
    this.name = "ApiError";
  }
}

export async function requestApi<T = any>(
  url: string,
  data: Record<string, unknown> = {}
): Promise<T> {
  const res = await fetchSyncPost(url, data);
  if (res.code !== 0) {
    showMessage(res.msg || `请求 ${url} 失败`, 5000, "error");
    throw new ApiError(res.code, res.msg, url);
  }
  return res.data as T;
}
```

## 2. 文档与内容块操作

### 2.1 创建文档（Markdown）

```ts
await requestApi("/api/filetree/createDocWithMd", {
  notebook: "20260913080000-nb12345",
  path: "/开发笔记/SiYuan插件开发实战",
  markdown: "# SiYuan 插件开发实战\n\n这是通过 API 创建的文档正文。"
});
```

### 2.2 插入块（段落、标题、自定义块）

```ts
// 插入普通 Markdown 段落
await requestApi("/api/block/insertBlock", {
  dataType: "markdown",
  data: "这是一个新插入的段落块。",
  parentID: "20260913080000-doc12345"
});

// 插入自定义块 (NodeCustomBlock)
await requestApi("/api/block/insertBlock", {
  dataType: "markdown",
  data: "```my-card\n自定义卡片内容数据\n```",
  previousID: "20260913080000-block12345"
});
```

### 2.3 获取块的 Kramdown 源码

```ts
const { kramdown } = await requestApi<{ kramdown: string }>("/api/block/getBlockKramdown", {
  id: "20260913080000-block12345"
});
console.log("Kramdown 原始内容:", kramdown);
```

### 2.4 全量更新块内容

```ts
await requestApi("/api/block/updateBlock", {
  id: "20260913080000-block12345",
  dataType: "markdown",
  data: "**更新后的粗体内容**"
});
```

## 3. 自定义属性（`custom-*`）

```ts
// 为指定块设置业务属性
await requestApi("/api/attr/setBlockAttrs", {
  id: "20260913080000-block12345",
  attrs: {
    "custom-task-status": "in_progress",
    "custom-assigned-user": "developer_alex",
    "custom-due-date": "2026-09-30"
  }
});

// 读取块的全部属性
const attrs = await requestApi<Record<string, string>>("/api/attr/getBlockAttrs", {
  id: "20260913080000-block12345"
});
console.log("自定义任务状态:", attrs["custom-task-status"]);
```

## 4. SQL 结构化检索

```ts
interface IBlockRow {
  id: string;
  type: string;
  subtype: string;
  content: string;
  updated: string;
}

const rows = await requestApi<IBlockRow[]>("/api/query/sql", {
  stmt: `
    SELECT id, type, subtype, content, updated
    FROM blocks
    WHERE type IN ('p', 'h', 'custom') AND content LIKE '%重要%'
    ORDER BY updated DESC
    LIMIT 20
  `
});
```

## 7. 资源上传与二进制文件

### 上传图片至指定笔记本资源库

```ts
export async function uploadAsset(file: File): Promise<string> {
  const formData = new FormData();
  formData.append("assetsDirPath", "/assets/");
  formData.append("file[]", file, file.name);

  const response = await fetch("/api/asset/upload", {
    method: "POST",
    body: formData,
  });
  const result = await response.json();
  if (result.code !== 0) throw new Error(result.msg);
  // 返回资源在工作空间内的引用路径
  return result.data.succMap[file.name];
}
```
